import numpy as np import matplotlib.pyplot as plt # ========================================== # 1. 시뮬레이션 기본 설정 및 입력 데이터 # ========================================== # 시간 설정 (초 단위) t_crit = 120.0 # 대기권 탈출 및 궤도 전환 임계 시간 (120초, 고도 약 100km) t_final = 600.0 # 목표 궤도 진입 시간 (600초) # [구간 1] 대기권 전용 기존 궤도 함수 (예: 중력 턴 비행 모델) # r_atm(t) = [x(t), y(t), z(t)] def get_atmospheric_trajectory(t): # 가상의 대기권 내 수직 및 경사 가속도 모델 x = 0.15 * t**1.8 y = 0.05 * t**1.5 z = 0.80 * t**1.9 return np.array([x, y, z]) def get_atmospheric_velocity(t, dt=1e-5): # 수치 미분을 통한 $t_{crit}$ 시점에서의 속도 벡터 산출 (C^1 연속성용) r_plus = get_atmospheric_trajectory(t + dt) r_minus = get_atmospheric_trajectory(t - dt) return (r_plus - r_minus) / (2 * dt) # 전환 시점(t_crit)에서의 경계 조건 (Boundary Conditions) r_crit = get_atmospheric_trajectory(t_crit) # d0: 위치 벡터 고정값 v_crit = get_atmospheric_velocity(t_crit) # d1: 속도 벡터 고정값 # 외기권에서 통신 최적화 목표가 될 위성들의 좌표 (아시아 상공 고정) # 시간 tau = t - t_crit (tau > 0) 기준 위성 목표점 4개 예시 tau_sat = np.array([60.0, 180.0, 300.0, 420.0]) # tau_1 ~ tau_4 sats_pos = np.array([ [2500.0, 800.0, 2100.0], # 위성 1 (km) [5200.0, 1800.0, 4100.0], # 위성 2 (km) [8100.0, 2900.0, 6000.0], # 위성 3 (km) [10500.0, 3800.0, 7200.0] # 위성 4 (km) ]) # ========================================== # 2. 제약조건이 반영된 최소제곱법 (구간 2 최적화) # ========================================== # 외기권 다항식: r_opt(tau) = d0 + d1*tau + d2*tau^2 + d3*tau^3 # 미지수: d2, d3 (각 축별 2개씩) # 변형된 목적 벡터: b_mod = sat_pos - (d0 + d1*tau) degree_free = 2 # 자율 미지수 계수 개수 (d2, d3 -> 2차, 3차 항) # 디자인 행렬 A_mod 구성 (tau^2, tau^3) A_mod = np.column_stack([tau_sat**2, tau_sat**3]) # 각 축(X, Y, Z)별 정규방정식을 통한 최적 계수 도출 d2_vec = np.zeros(3) d3_vec = np.zeros(3) for axis in range(3): d0 = r_crit[axis] d1 = v_crit[axis] # 위치/속도 제약조건을 뺀 변형된 우변 벡터 b_mod = sats_pos[:, axis] - (d0 + d1 * tau_sat) # 정규방정식 풀이: (A^T * A) * c = A^T * b coeffs_mod, residuals, rank, s = np.linalg.lstsq(A_mod, b_mod, rcond=None) d2_vec[axis] = coeffs_mod[0] d3_vec[axis] = coeffs_mod[1] # ========================================== # 3. 이중 궤도(Dual-Phase) 연속 복원 및 시각화 # ========================================== # 타임라인 생성 t_phase1 = np.linspace(0, t_crit, 100) t_phase2 = np.linspace(t_crit, t_final, 200) # 구간 1 궤도 계산 traj_phase1 = np.array([get_atmospheric_trajectory(t) for t in t_phase1]) # 구간 2 최적화 궤도 계산 tau_phase2 = t_phase2 - t_crit traj_phase2 = np.zeros((len(t_phase2), 3)) for i, tau in enumerate(tau_phase2): # r_opt(tau) = d0 + d1*tau + d2*tau^2 + d3*tau^3 pos = r_crit + v_crit * tau + d2_vec * (tau**2) + d3_vec * (tau**3) traj_phase2[i] = pos # 전체 이중 궤도 병합 full_trajectory = np.vstack((traj_phase1, traj_phase2)) full_time = np.concatenate((t_phase1, t_phase2)) # ========================================== # 4. 결과 출력 및 수치 검증 # ========================================== print("=== 이중 궤도(Dual-Phase) 최소제곱법 도출 결과 ===") print(f"1. 전환 지점 (t = {t_crit}s) 위치 (km): X={r_crit[0]:.2f}, Y={r_crit[1]:.2f}, Z={r_crit[2]:.2f}") print(f"2. 전환 지점 (t = {t_crit}s) 속도 (km/s): Vx={v_crit[0]:.3f}, Vy={v_crit[1]:.3f}, Vz={v_crit[2]:.3f}") print("-" * 50) print(f"3. 외기권 최적화 다항식 계수 (X축): d0={r_crit[0]:.2f}, d1={v_crit[0]:.4f}, d2={d2_vec[0]:.6e}, d3={d3_vec[0]:.6e}") print(f"4. C0(위치) 연속성 확인: Phase1 끝점={traj_phase1[-1, 0]:.2f} km | Phase2 시작점={traj_phase2[0, 0]:.2f} km") # Matplotlib 3D 시각화 fig = plt.figure(figsize=(10, 7)) ax = fig.add_subplot(111, projection='3d') # 구간 1 (대기권 기존 궤도) ax.plot(traj_phase1[:, 0], traj_phase1[:, 1], traj_phase1[:, 2], color='orange', linewidth=2.5, label='Phase 1: Atmospheric (Gravity Turn)') # 구간 2 (외기권 최소제곱 최적화 궤도) ax.plot(traj_phase2[:, 0], traj_phase2[:, 1], traj_phase2[:, 2], color='deepskyblue', linewidth=2.5, label='Phase 2: Space Least-Squares Opt') # 위성 목표점 ax.scatter(sats_pos[:, 0], sats_pos[:, 1], sats_pos[:, 2], color='red', s=60, marker='^', label='Asia Satellites') # 전환점 (t_crit) ax.scatter(r_crit[0], r_crit[1], r_crit[2], color='green', s=80, marker='o', label='Transition Point (t_crit)') ax.set_title('Dual-Phase Trajectory Optimization via Least Squares') ax.set_xlabel('X (km)') ax.set_ylabel('Y (km)') ax.set_zlabel('Z (km)') ax.legend() plt.tight_layout() plt.show()
